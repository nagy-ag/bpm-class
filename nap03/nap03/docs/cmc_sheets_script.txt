// Sheets > Extensions > Apps Script. Az egész fájlt másold be.
// Az ARFOLYAMOK munkalapot ez a függvény használja.
function arfolyamFrissites() {
  const API_KEY = 'IDE_A_SAJAT_KULCSOD';
  const IDS = ['1', '1027']; // Harmadik eszközhöz csak ezt a listát bővítsd.
  if (API_KEY === 'IDE_A_SAJAT_KULCSOD' || !API_KEY.trim()) {
    throw new Error('Előbb írd be a saját CMC-kulcsodat az idézőjelek közé.');
  }
  const url = 'https://pro-api.coinmarketcap.com/v1/cryptocurrency/quotes/latest'
    + '?id=' + IDS.join(',') + '&convert=EUR&CMC_PRO_API_KEY=' + encodeURIComponent(API_KEY.trim());
  const response = UrlFetchApp.fetch(url, {muteHttpExceptions: true});
  let result;
  try { result = JSON.parse(response.getContentText()); }
  catch (e) { throw new Error('Nem JSON-válasz érkezett. HTTP: ' + response.getResponseCode()); }
  if (response.getResponseCode() !== 200 || !result.status || result.status.error_code !== 0) {
    throw new Error('CMC-hiba: ' + (result.status && result.status.error_message || response.getResponseCode()));
  }
  const now = new Date();
  const rows = [['Kripto', 'Ár (EUR)', 'Frissítve']];
  IDS.forEach(id => {
    const coin = result.data && result.data[id];
    const price = coin && coin.quote && coin.quote.EUR && coin.quote.EUR.price;
    if (typeof price !== 'number') throw new Error('Hiányzó vagy hibás ár. CMC-azonosító: ' + id);
    rows.push([coin.name, price, now]);
  });
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error('A kódot a táblázat Extensions > Apps Script menüjéből futtasd.');
  const sheet = ss.getSheetByName('ARFOLYAMOK') || ss.insertSheet('ARFOLYAMOK');
  sheet.getRange('A:C').clearContent();
  sheet.getRange(1, 1, rows.length, 3).setValues(rows);
  sheet.getRange(2, 2, IDS.length, 1).setNumberFormat('0.00');
  sheet.getRange(2, 3, IDS.length, 1).setNumberFormat('yyyy-mm-dd hh:mm:ss');
}
